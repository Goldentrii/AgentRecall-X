# Cards v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `session_start` shows the owner identity card first and the project state card after HARD RULES. `session_end` rewrites the state card. CLI hooks use the real session id.
**Architecture:** Three core storage modules: `hook-session.ts`, `identity-card.ts`, `state-card.ts`. Each holds pure functions plus readers that never throw. `sessionStart` and `sessionEnd` call them. The MCP and CLI renderers share `formatOwnerSection` and `formatStateSection`, and only the CLI writes the identity card.
**Tech Stack:** TypeScript (Node16 ESM), node:test `.mjs`, zod v4, npm workspaces.
**Spec:** docs/superpowers/specs/2026-10-31-cards-v1-design.md

## Global Constraints

- Identity body is capped at 1,600 chars (a longer write is rejected). The whole state card file is capped at 2,400 chars.
- State field caps: goal 300, next 300, open_decisions 5×160, blockers 5×160, pointers 8×200. Overflow is cut to `…` and reported.
- No version bump, publish or push. Changelog edits go under `[Unreleased]` only.
- The identity card is personal tier and never synced. Do not touch `sync.ts`.
- Card writes use a unique tmp file plus `renameAtomic`.
- Tests: node:test `.mjs` with temp roots. Leave the shell's `AGENT_RECALL_ROOT` unset.
- Windows pointers must round-trip unchanged. Do not change `handoff.md`.
- **Build before test.** Every test imports `dist/`. Core tests use `../dist/index.js`, or `agent-recall-core`, which is a symlink to `packages/core` and resolves to its `dist`. MCP and SDK tests use `../dist`, and CLI tests spawn `dist/index.js`. So `npm run build -w packages/core` must run before `npm test -w packages/core`. Downstream packages need core built first, then their own build.

## Review Focus

/** Cards v1 (G1) — one session-id resolver for every CLI hook. Core getSessionId() is unchanged. */
import type { Readable } from "node:stream";

const ENV_KEYS = ["CLAUDE_CODE_SESSION_ID", "CLAUDE_SESSION_ID", "SESSION_ID"] as const;
const clean = (v: unknown): string | undefined =>
  typeof v === "number" && Number.isFinite(v) ? String(v) : typeof v === "string" && v.trim() ? v.trim() : undefined;

export function resolveHookSessionId(stdinPayload: Record<string, unknown> | undefined, env: NodeJS.ProcessEnv = process.env): string | undefined {
  for (const v of [stdinPayload?.["session_id"], ...ENV_KEYS.map((k) => env[k])]) {
    const s = clean(v);
    if (s) return s;
  }
  return undefined;
}

export function hookLockKey(sid: string | undefined, now: Date = new Date()): string {
  return sid ?? `nosid-${now.toISOString().slice(0, 13)}`;
}

/** TTY → {} at once; a stream that never ends → {} after timeoutMs (stream destroyed). Invalid/non-object JSON → {}. */
export function readHookStdin(stream: Readable & { isTTY?: boolean } = process.stdin, timeoutMs = 2000): Promise<Record<string, unknown>> {
  if (stream.isTTY) return Promise.resolve({});
  return new Promise((resolve) => {
    const chunks: Buffer[] = [];
    let timer: NodeJS.Timeout | undefined;
    let done = false;
    const onData = (c: Buffer | string) => { chunks.push(typeof c === "string" ? Buffer.from(c, "utf-8") : c); };
    const finish = (timedOut: boolean) => {
      if (done) return;
      done = true;
      clearTimeout(timer);
      stream.off("data", onData);
      if (timedOut) { try { stream.destroy(); } catch { /* best effort */ } }
      try {
        const v: unknown = JSON.parse(Buffer.concat(chunks).toString("utf-8").trim() || "{}");
        resolve(v !== null && typeof v === "object" && !Array.isArray(v) ? (v as Record<string, unknown>) : {});
      } catch { resolve({}); }
    };
    stream.on("data", onData);
    stream.once("end", () => finish(false));
    stream.once("error", () => finish(false));
    timer = setTimeout(() => finish(true), timeoutMs);
  });
}

/** Cards v1 — unique tmp (pid+random) so concurrent writers never share one; tmp removed if the rename fails. */
export function writeTextAtomicPrivate(filePath: string, text: string): void {
  ensureDir(path.dirname(filePath));
  const tmp = `${filePath}.tmp.${process.pid}.${randomBytes(4).toString("hex")}`;
  fs.writeFileSync(tmp, text, { encoding: "utf-8", mode: 0o600 });
  try { renameAtomic(tmp, filePath); } catch (err) { try { fs.rmSync(tmp, { force: true }); } catch { /* best effort */ } throw err; }
}

/** Cards v1 (G2) — <root>/identity-card.md. Personal tier, never synced; the CLI is the only writer. Reads never throw. */
import * as fs from "node:fs";
import * as path from "node:path";
import { getRoot } from "../types.js";
import { writeTextAtomicPrivate } from "./fs-utils.js";
import { scrubSecretContent } from "./content-guard.js";

export const IDENTITY_BODY_CAP = 1600;
export interface IdentityCard { version: number; updated: string; body: string }
export class IdentityCardError extends Error {
  constructor(public readonly code: "empty" | "too_long" | "secret", message: string) { super(message); this.name = "IdentityCardError"; }
}
export const identityCardPath = (): string => path.join(getRoot(), "identity-card.md");
const warn = (msg: string) => { try { process.stderr.write(`[AgentRecall] identity-card.md: ${msg}\n`); } catch { /* never throw */ } };

export function parseIdentityCard(text: string): { card: IdentityCard; malformed: boolean } {
  const t = text.replace(/\r\n/g, "\n");
  const m = /^---\n([\s\S]*?)\n---(?:\n|$)([\s\S]*)$/.exec(t);
  const fm = new Map((m?.[1] ?? "").split("\n").map((l) => [l.split(":")[0].trim(), l.slice(l.indexOf(":") + 1).replace(/\s+#.*$/, "").trim()] as [string, string]));
  const version = Number(fm.get("version"));
  if (m && fm.get("kind") === "identity-card" && Number.isInteger(version) && version >= 0) {
    return { card: { version, updated: fm.get("updated") ?? "", body: m[2].replace(/^\n+/, "").trimEnd() }, malformed: false };
  }
  return { card: { version: 0, updated: "", body: t.trimEnd() }, malformed: true };
}

export function readIdentityCard(): IdentityCard | null {
  let raw: string;
  try {
    if (!fs.existsSync(identityCardPath())) return null;
    raw = fs.readFileSync(identityCardPath(), "utf-8");
  } catch (err) { warn(`unreadable (${String(err)}) — skipped`); return null; }
  if (raw.includes("\u0000")) { warn("binary content — skipped"); return null; }
  if (!raw.trim()) { warn("empty file — skipped"); return null; }
  const { card, malformed } = parseIdentityCard(raw);
  if (malformed) warn("malformed frontmatter — whole file used as body");
  else if (card.body.length > IDENTITY_BODY_CAP) warn(`body over ${IDENTITY_BODY_CAP} chars — truncated for this load`);
  if (card.body.length > IDENTITY_BODY_CAP) card.body = card.body.slice(0, IDENTITY_BODY_CAP - 1) + "…";
  return card;
}

export function writeIdentityCard(body: string, now: Date = new Date()): IdentityCard & { path: string } {
  const clean = body.replace(/\r\n/g, "\n").trim();
  if (!clean) throw new IdentityCardError("empty", "identity card body is empty — nothing written");
  if (clean.length > IDENTITY_BODY_CAP) throw new IdentityCardError("too_long", `identity card body is ${clean.length} chars; cap is ${IDENTITY_BODY_CAP} — nothing written`);
  const scan = scrubSecretContent(clean);
  if (scan.redactedCount > 0) throw new IdentityCardError("secret", `identity card body contains a secret (${scan.labels[0] ?? "secret pattern"}) — nothing written`);
  const card = { version: (readIdentityCard()?.version ?? 0) + 1, updated: now.toISOString(), body: clean };
  writeTextAtomicPrivate(identityCardPath(), `---\nkind: identity-card\nversion: ${card.version}\nupdated: ${card.updated}\n---\n${clean}\n`);
  return { ...card, path: identityCardPath() };
}

/** Shared renderer (MCP terse/verbose, CLI hook-start). */
export const formatOwnerSection = (o: { version: number; body: string }): string[] => [`Owner (identity card v${o.version})`, o.body];

/** Cards v1 (G4) — projects/<slug>/state-card.md, REWRITTEN at session_end with carry-forward. Values single-line (no forged "## "), scrubbed, capped per field and per file; age_days computed at read time. */
import * as fs from "node:fs";
import { projectSubPath } from "./paths.js";
import { writeTextAtomicPrivate } from "./fs-utils.js";
import { scrubForCloud } from "./content-guard.js";

export const STATE_CARD_FILE_CAP = 2400;
const CAP = { goal: 300, next: 300, open_decisions: 5, blockers: 5, pointers: 8, item: 160, pointer: 200, summary: 200 } as const;
export interface StateCardInput { goal?: string; open_decisions?: string[]; blockers?: string[]; next?: string; pointers?: string[] }
export interface StateCardBody { updated: string; session: string; goal: string; open_decisions: string[]; blockers: string[]; next: string; pointers: string[]; last_session: string }
export interface StateCard extends StateCardBody { age_days: number }
export interface MergeStateCardArgs { previous: StateCardBody | null; state?: StateCardInput; trajectory?: string; summary?: string; session: string; now?: Date }
type List = "open_decisions" | "blockers" | "pointers";
const LISTS: string[] = ["open_decisions", "blockers", "pointers"];
const SECTIONS: Array<[string, keyof StateCardBody]> = [["Goal", "goal"], ["Open decisions", "open_decisions"], ["Blockers", "blockers"], ["Next", "next"], ["Pointers", "pointers"], ["Last session", "last_session"]];
const oneLine = (s: string) => s.replace(/\s+/g, " ").trim();
const warn = (slug: string, msg: string) => { try { process.stderr.write(`[AgentRecall] ${slug}/state-card.md: ${msg}\n`); } catch { /* never throw */ } };
export const stateCardPath = (slug: string): string => projectSubPath(slug, "state-card.md");

export function serializeStateCard(c: StateCardBody): string {
  const out = ["---", "kind: state-card", `updated: ${c.updated}`, `session: ${c.session}`, "---"];
  for (const [h, k] of SECTIONS) {
    const v = c[k];
    if (v.length) out.push(`## ${h}`, ...(Array.isArray(v) ? v.map((x) => `- ${x}`) : [v]));
  }
  return out.join("\n") + "\n";
}

/** Normalize + cap each field, then pop list tails (longest first; stable ties pointers → open_decisions → blockers) until the file fits. Idempotent. */
export function capStateCard(c: StateCardBody): { card: StateCardBody; truncated: string[] } {
  const cut = new Set<string>();
  const clip = (name: string, s: string, cap: number) => {
    const t = oneLine(scrubForCloud(s));
    if (t.length <= cap) return t;
    if (name) cut.add(name);
    return t.slice(0, cap - 1) + "…";
  };
  const list = (name: List) => {
    const all = c[name].map(oneLine).filter((x) => x && x !== "(none)");
    if (all.length > CAP[name]) cut.add(name);
    return all.slice(0, CAP[name]).map((x) => clip(name, x, name === "pointers" ? CAP.pointer : CAP.item));
  };
  const card: StateCardBody = {
    updated: c.updated, session: c.session.replace(/\s+/g, "").slice(0, 200),
    goal: clip("goal", c.goal, CAP.goal), next: clip("next", c.next, CAP.next),
    open_decisions: list("open_decisions"), blockers: list("blockers"), pointers: list("pointers"),
    last_session: clip("", c.last_session, 13 + CAP.summary),
  };
  while (serializeStateCard(card).length > STATE_CARD_FILE_CAP) {
    const v = (["pointers", "open_decisions", "blockers"] as const).filter((f) => card[f].length).sort((a, b) => card[b].length - card[a].length)[0];
    if (!v) break;
    card[v] = card[v].slice(0, -1);
    cut.add(v);
  }
  return { card, truncated: [...cut] };
}

export function mergeStateCard(a: MergeStateCardArgs): { card: StateCardBody; truncated: string[] } | null {
  const p = a.previous, s: StateCardInput = a.state ?? {}, summary = oneLine(a.summary ?? ""), now = a.now ?? new Date();
  if (!p && !a.state && !a.trajectory && !summary) return null;
  return capStateCard({
    updated: now.toISOString(), session: a.session,
    goal: s.goal ?? p?.goal ?? "", next: s.next ?? (a.trajectory || p?.next || ""),
    open_decisions: s.open_decisions ?? p?.open_decisions ?? [], blockers: s.blockers ?? p?.blockers ?? [], pointers: s.pointers ?? p?.pointers ?? [],
    last_session: summary ? `${now.toLocaleDateString("sv")} — ${summary.slice(0, CAP.summary)}` : p?.last_session ?? "",
  });
}

export function parseStateCard(text: string, now: Date = new Date()): StateCard | null {
  const m = /^---\n([\s\S]*?)\n---(?:\n|$)([\s\S]*)$/.exec(text.replace(/\r\n/g, "\n"));
  const fm = new Map((m?.[1] ?? "").split("\n").map((l) => [l.split(":")[0].trim(), l.slice(l.indexOf(":") + 1).trim()] as [string, string]));
  const ms = Date.parse(fm.get("updated") ?? "");
  if (!m || fm.get("kind") !== "state-card" || Number.isNaN(ms)) return null;
  const v = new Map<string, string[]>();
  let cur: string | undefined;
  for (const line of m[2].split("\n")) {
    const h = /^## (.+?)\s*$/.exec(line);
    if (h) cur = SECTIONS.find(([name]) => name === h[1])?.[1];
    else if (cur && line.trim() && (!LISTS.includes(cur) || line.startsWith("- "))) v.set(cur, [...(v.get(cur) ?? []), LISTS.includes(cur) ? line.slice(2) : line.trim()]);
  }
  const join = (k: string) => (v.get(k) ?? []).join(" ");
  const { card } = capStateCard({ updated: fm.get("updated") ?? "", session: fm.get("session") ?? "", goal: join("goal"), next: join("next"), last_session: join("last_session"), open_decisions: v.get("open_decisions") ?? [], blockers: v.get("blockers") ?? [], pointers: v.get("pointers") ?? [] });
  return { ...card, age_days: Math.max(0, Math.floor((now.getTime() - ms) / 86_400_000)) };
}

export function readStateCard(slug: string, now: Date = new Date()): StateCard | null {
  try {
    if (!fs.existsSync(stateCardPath(slug))) return null;
    const raw = fs.readFileSync(stateCardPath(slug), "utf-8");
    if (raw.includes("\u0000")) { warn(slug, "binary content — skipped"); return null; }
    const card = parseStateCard(raw, now);
    if (!card) warn(slug, raw.trim() ? "malformed frontmatter — skipped" : "empty file — skipped");
    return card;
  } catch (err) { warn(slug, `unreadable (${String(err)}) — skipped`); return null; }
}

export function writeStateCard(slug: string, card: StateCardBody): { path: string; chars: number; truncated: string[] } {
  const { card: capped, truncated } = capStateCard(card);
  const text = serializeStateCard(capped);
  writeTextAtomicPrivate(stateCardPath(slug), text);
  return { path: stateCardPath(slug), chars: text.length, truncated };
}

/** Shared renderer: non-empty fields in the order Goal, Blockers, Next, Open decisions, Pointers, Last session. */
export function formatStateSection(c: StateCard): string[] {
  const d = c.updated.slice(0, 10);
  const out = [c.age_days >= 7 ? `State (updated ${d}, stale: ${c.age_days} d)` : `State (updated ${d})`];
  const list = (label: string, v: string[]) => { if (v.length) out.push(`${label}:`, ...v.map((x) => `  - ${x}`)); };
  if (c.goal) out.push(`Goal: ${c.goal}`);
  list("Blockers", c.blockers);
  if (c.next) out.push(`Next: ${c.next}`);
  list("Open decisions", c.open_decisions);
  list("Pointers", c.pointers);
  if (c.last_session) out.push(`Last session: ${c.last_session}`);
  return out;
}
